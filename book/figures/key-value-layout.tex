\small
假设有 $M=3$ 条 Key，每条 Key 有 $d_k=2$ 个数字。
\textbf{相同底色表示同一条 Key：两栏中的数字相同，只是摆放方向不同。}

\vspace{1.5mm}
\noindent
\begin{minipage}[t]{.49\linewidth}

$K\in\R^{M\times d_k}$\\
{\footnotesize 每行一条 Key}

\end{tikzpicture}\\[1mm]
$K^\top\in\R^{d_k\times M}$\\
{\footnotesize 每列一条 Key}
\end{minipage}

\begin{align*}
\underbrace{K}_{M\times d_k}
\underbrace{\vect q}_{d_k\times1}
&=\underbrace{\vect s}_{M\times1},\\
\underbrace{\vect q^\top}_{1\times d_k}
\underbrace{K^\top}_{d_k\times M}
&=\underbrace{\vect s^\top}_{1\times M}.
\end{align*}
\begin{center}
{\color{blue}\bfseries $(K\vect q)^\top=\vect q^\top K^\top$}
\end{center}

\textbf{Value 同理：}约定 $V$ 的\textbf{每行是一条 Value}，
可以按需转置，读取相同的加权结果：
\[
\underbrace{V}_{M\times d_v}
\quad\xleftrightarrow{\text{转置}}\quad
\underbrace{V^\top}_{d_v\times M}.
\]
\begin{align*}
\underbrace{V^\top}_{d_v\times M}
\underbrace{\vect a}_{M\times1}
&=\underbrace{\vect y}_{d_v\times1},\\
\underbrace{\vect a^\top}_{1\times M}
\underbrace{V}_{M\times d_v}
&=\underbrace{\vect y^\top}_{1\times d_v}.
\end{align*}
\textbf{检查：}若 $M=3$、$d_k=2$、$d_v=4$，
则 $K:3\times2$、$K^\top:2\times3$、
$V:3\times4$、$V^\top:4\times3$。
\textbf{Key 和 Value 不必具有相同维度。}

\textbf{重要边界：}上面比较的是\emph{同一个单 Query}的转置写法。
扩展成多个 Query 后使用 $QK^\top$；
不要把 $QK^\top$ 和 $KQ^\top$ 直接当作可以交换的运算。
